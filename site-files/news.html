  <!-- Dynamic Fetch Script -->

  <script>

    document.addEventListener('DOMContentLoaded', async function () {

      const container = document.getElementById('news-grid');

      if (!container) return;


      try {

        const response = await fetch('/api/news?limit=30');

        if (!response.ok) throw new Error('Network response was not ok');

        

        const data = await response.json();

        if (!data.articles || data.articles.length === 0) return;


        // Clear existing static fallback cards

        container.innerHTML = '';


        // Build real cards from API database results

        data.articles.forEach(article => {

          const card = document.createElement('article');

          card.className = 'news-card';


          const articleLink = `/news-article.html?slug=${encodeURIComponent(article.slug)}`;

          const imageSrc = article.image_url || 'news-placeholder.jpg';


          card.innerHTML = `

            <a href="${articleLink}" style="text-decoration:none; color:inherit; display:flex; flex-direction:column; height:100%;">

              <img src="${imageSrc}" alt="${article.title}" class="news-card-image" loading="lazy" />

              <div class="news-card-content">

                <span class="news-card-date">${article.date_label || ''}</span>

                <h2 class="news-card-title">${article.title}</h2>

                <p class="news-card-snippet">${article.summary || ''}</p>

                <span class="news-card-link">Read Article &rarr;</span>

              </div>

            </a>

          `;

          container.appendChild(card);

        });

      } catch (err) {

        console.error('Failed to load real-time news:', err);

      }

    });

  </script>
