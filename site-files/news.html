  <!-- Live Dynamic Rendering Script -->

  <script>

    document.addEventListener('DOMContentLoaded', function () {

      const grid = document.getElementById('news-grid');

      if (!grid) return;


      fetch('/api/news?limit=30')

        .then(function (res) {

          if (!res.ok) throw new Error('HTTP ' + res.status);

          return res.json();

        })

        .then(function (data) {

          if (!data.articles || data.articles.length === 0) {

            grid.innerHTML = '<p style="grid-column: 1/-1; text-align: center;">No articles published yet.</p>';

            return;

          }


          grid.innerHTML = ''; // Clear loading text


          data.articles.forEach(function (art) {

            const card = document.createElement('article');

            card.className = 'news-card';


            const title = art.title || 'Untitled Article';

            const snippet = art.summary || art.highlight || art.body || '';

            const image = art.image_url || 'hero-bg.jpg';

            const date = art.date_label || art.published_date || '';

            const link = art.slug ? `/news-article.html?slug=${encodeURIComponent(art.slug)}` : '#';


            card.innerHTML = `

              <a href="${link}" style="text-decoration: none; color: inherit; display: flex; flex-direction: column; height: 100%;">

                <img src="${image}" alt="${title}" class="news-card-image" onerror="this.src='hero-bg.jpg'" />

                <div class="news-card-content">

                  <span class="news-card-date">${date}</span>

                  <h2 class="news-card-title">${title}</h2>

                  <p class="news-card-snippet">${snippet.substring(0, 160)}${snippet.length > 160 ? '...' : ''}</p>

                  <span class="news-card-link">Read Full Article &rarr;</span>

                </div>

              </a>

            `;

            grid.appendChild(card);

          });

        })

        .catch(function (err) {

          console.error('Failed to load news:', err);

          grid.innerHTML = '<p style="grid-column: 1/-1; text-align: center; color: #ef4444;">Unable to load news right now. Please try refreshing.</p>';

        });

    });

  </script>
